# keepsake v1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A Python MCP server that stores OKF v0.2 concepts in Postgres, enforces tenant isolation with row-level security, and serves agents seven bounded tools.

**Architecture:** Three layers. `okf_core` is a pure library over bytes and dataclasses with no database dependency. `keepsake.store` owns all SQL and sets the tenant scope per transaction. `keepsake.server` exposes MCP tools over HTTP. Postgres is the only store; there is no materialised bundle and no mount.

**Tech Stack:** Python 3.14, uv, psycopg 3 (+ `psycopg_pool`), Alembic, `ruamel.yaml`, the official `mcp` SDK, pytest, testcontainers, ruff, ty.

**Spec:** `docs/design.md`

## Global Constraints

- Python 3.14. All tooling invoked via `uv run`.
- The GUC is `okf.current_tenant`. Never `app.current_tenant`.
- Scope is set with `SET LOCAL` inside an explicit transaction. Never `SET`.
- All RLS policies carry both `USING` and `WITH CHECK`.
- All tables are `ENABLE` **and** `FORCE ROW LEVEL SECURITY`.
- Every database object is schema-qualified. Default schema `okf`, configurable.
- `to_tsvector` is always called with the two-argument `(regconfig, text)` form.
- Search terms are OR-ed, never AND-ed.
- `limit` is a required parameter on `okf_search` and `okf_grep`.
- `okf_search` never returns a concept body.
- Error messages and logs MUST NOT contain concept bodies.
- `okf_core` MUST NOT import anything under `keepsake`.

`src/okf_core/frontmatter.py`:

```python
"""OKF frontmatter split and round-trip. ruamel preserves key order and style."""

from __future__ import annotations

import io
import re
from typing import Any

from ruamel.yaml import YAML

_FENCE = re.compile(r"\A---\n(.*?)\n---\n(.*)\Z", re.DOTALL)
_KNOWN = ("type", "title", "description")

_yaml = YAML()
_yaml.preserve_quotes = True


def split(text: str) -> tuple[dict[str, Any], str]:
    """Return (frontmatter mapping, body). A document without a fence has no frontmatter."""
    if not (m := _FENCE.match(text)):
        return {}, text
    return dict(_yaml.load(m.group(1)) or {}), m.group(2)


def join(meta: dict[str, Any], body: str) -> str:
    buf = io.StringIO()
    _yaml.dump(meta, buf)
    return f"---\n{buf.getvalue()}---\n{body}"
```

`src/okf_core/__init__.py`:

```python
from __future__ import annotations

from dataclasses import dataclass, field
from typing import Any

from okf_core.frontmatter import join, split
from okf_core.links import extract_links


@dataclass(frozen=True, slots=True)
class Concept:
    path: str
    type: str
    title: str = ""
    description: str = ""
    body: str = ""
    frontmatter: dict[str, Any] = field(default_factory=dict)
    links: tuple[str, ...] = ()
    version: int = 1


def parse(text: str, path: str) -> Concept:
    meta, body = split(text)
    known = {k: str(meta.pop(k, "")) for k in ("type", "title", "description")}
    return Concept(
        path=path,
        body=body,
        frontmatter=meta,
        links=extract_links(body),
        **known,
    )


def serialize(c: Concept) -> str:
    meta: dict[str, Any] = {"type": c.type}
    if c.title:
        meta["title"] = c.title
    if c.description:
        meta["description"] = c.description
    meta.update(c.frontmatter)
    return join(meta, c.body)
```

`src/okf_core/links.py`:

```python
"""Outbound edges are markdown links to other concepts. External URLs are not edges."""

from __future__ import annotations

import re

_LINK = re.compile(r"\[[^\]]*\]\(([^)\s]+)\)")
_EXTERNAL = re.compile(r"\A[a-z][a-z0-9+.-]*://|\A(?:mailto|tel):", re.IGNORECASE)


def extract_links(body: str) -> tuple[str, ...]:
    seen: dict[str, None] = {}
    for target in _LINK.findall(body):
        if _EXTERNAL.match(target):
            continue
        seen.setdefault(target.removesuffix(".md"), None)
    return tuple(seen)
```

`src/okf_core/validate.py`:

```python
"""Per-write validation. Bundle-level link resolution lives in the CLI, not here."""

from __future__ import annotations

from okf_core import Concept


def validate(c: Concept) -> list[str]:
    errors: list[str] = []
    if not c.type.strip():
        errors.append("type is required")
    if c.path.startswith("/"):
        errors.append("path must be relative, not absolute")
    elif ".." in c.path.split("/"):
        errors.append("path must not traverse upward")
    elif not c.path.strip():
        errors.append("path is required")
    return errors
```

`src/keepsake/store/pool.py`:

```python
"""Connection pooling and tenant scoping. The GUC is set here and nowhere else."""

from __future__ import annotations

from collections.abc import Iterator
from contextlib import contextmanager
from uuid import UUID

import psycopg
from psycopg_pool import ConnectionPool


class Store:
    def __init__(self, dsn: str, schema: str = "okf") -> None:
        self._schema = schema
        self._pool = ConnectionPool(dsn, open=True, kwargs={"autocommit": False})

    @contextmanager
    def raw(self) -> Iterator[psycopg.Connection]:
        """A connection with no tenant scope. For startup checks only."""
        with self._pool.connection() as conn:
            conn.execute(f"SET LOCAL search_path = {self._schema}, pg_catalog")
            yield conn

    @contextmanager
    def scope(self, tenant_id: UUID) -> Iterator[psycopg.Connection]:
        with self._pool.connection() as conn:
            # SET LOCAL, never SET: a session-scoped value would be inherited by
            # the next request to reuse this pooled connection.
            conn.execute(f"SET LOCAL search_path = {self._schema}, pg_catalog")
            conn.execute("SET LOCAL okf.current_tenant = %s", (str(tenant_id),))
            yield conn
```

```python
"""Concept persistence. Two statements, never one upsert: an upsert guarded by a
version predicate silently creates a row the caller believed it was updating."""

from __future__ import annotations

import json
from dataclasses import asdict, dataclass
from uuid import UUID

from okf_core import Concept
from keepsake.store.pool import Store

_COLS = "tenant_id, path, type, title, description, body, frontmatter, links, updated_by"


@dataclass(frozen=True, slots=True)
class Conflict:
    current_version: int
    current_body: str


class ConceptStore:
    def __init__(self, store: Store) -> None:
        self._store = store

    def create(self, tenant_id: UUID, c: Concept, actor: str) -> int | None:
        with self._store.scope(tenant_id) as conn:
            row = conn.execute(
                f"INSERT INTO okf.concept ({_COLS}) "
                "VALUES (%s,%s,%s,%s,%s,%s,%s,%s,%s) "
                "ON CONFLICT (tenant_id, path) DO NOTHING RETURNING version",
                (tenant_id, c.path, c.type, c.title, c.description, c.body,
                 json.dumps(c.frontmatter), list(c.links), actor),
            ).fetchone()
            if row is None:
                return None
            self._revise(conn, tenant_id, c, row[0], "create", actor)
            return int(row[0])

    def update(
        self, tenant_id: UUID, c: Concept, actor: str, expected_version: int | None
    ) -> int | Conflict:
        with self._store.scope(tenant_id) as conn:
            row = conn.execute(
                "UPDATE okf.concept SET type=%s, title=%s, description=%s, body=%s, "
                "frontmatter=%s, links=%s, updated_by=%s, version=version+1, updated_at=now() "
                "WHERE path=%s AND (%s::int IS NULL OR version=%s) RETURNING version",
                (c.type, c.title, c.description, c.body, json.dumps(c.frontmatter),
                 list(c.links), actor, c.path, expected_version, expected_version),
            ).fetchone()
            if row is not None:
                self._revise(conn, tenant_id, c, row[0], "update", actor)
                return int(row[0])
            current = conn.execute(
                "SELECT version, body FROM okf.concept WHERE path=%s", (c.path,)
            ).fetchone()
            if current is None:
                raise KeyError(c.path)
            return Conflict(current_version=int(current[0]), current_body=str(current[1]))

    @staticmethod
    def _revise(conn, tenant_id: UUID, c: Concept, version: int, op: str, actor: str) -> None:
        snapshot = asdict(c) | {"version": version}
        snapshot["links"] = list(c.links)
        conn.execute(
            "INSERT INTO okf.concept_revision "
            "(tenant_id, path, version, op, snapshot, updated_by) VALUES (%s,%s,%s,%s,%s,%s)",
            (tenant_id, c.path, version, op, json.dumps(snapshot), actor),
        )
```

```python
"""Refuse to start when the database configuration silently disables isolation."""

from __future__ import annotations

from keepsake.store.pool import Store


class MisconfiguredDatabase(RuntimeError):
    """Raised at startup. Crashing loudly beats serving cross-tenant reads."""


def verify(store: Store, schema: str = "okf") -> None:
    with store.raw() as conn:
        if conn.execute(
            "SELECT usesuper FROM pg_user WHERE usename = current_user"
        ).fetchone()[0]:
            raise MisconfiguredDatabase(
                "okf must not connect as a superuser: RLS does not apply to superusers"
            )

        owner = conn.execute(
            "SELECT nspowner::regrole::text FROM pg_namespace WHERE nspname = %s",
            (schema,),
        ).fetchone()
        if owner and owner[0] == conn.execute("SELECT current_user").fetchone()[0]:
            raise MisconfiguredDatabase(
                f"okf must not connect as the role that owns the schema {schema}: "
                "a table owner bypasses row-level security"
            )

        for name, enabled, forced in conn.execute(
            "SELECT relname, relrowsecurity, relforcerowsecurity FROM pg_class c "
            "JOIN pg_namespace n ON n.oid = c.relnamespace "
            "WHERE n.nspname = %s AND c.relkind = 'r'", (schema,),
        ).fetchall():
            if not enabled:
                raise MisconfiguredDatabase(f"{schema}.{name} has row-level security disabled")
            if not forced:
                raise MisconfiguredDatabase(f"{schema}.{name} does not FORCE row-level security")
```

```python
"""The seven MCP tools. Validation runs here so every write path shares it."""

from __future__ import annotations

from typing import Any
from uuid import UUID

from okf_core import Concept, validate
from okf_core.links import extract_links
from keepsake.store.concepts import ConceptStore, Conflict


class ToolError(ValueError):
    """Surfaced to the agent. MUST NOT contain a concept body."""


class Tools:
    def __init__(self, concepts: ConceptStore, tenant_id: UUID, actor: str) -> None:
        self._c = concepts
        self._t = tenant_id
        self._actor = actor

    def _concept(self, path: str, **kw: Any) -> Concept:
        body = str(kw.get("body", ""))
        c = Concept(
            path=path, type=str(kw.get("type", "")), title=str(kw.get("title", "")),
            description=str(kw.get("description", "")), body=body,
            frontmatter=dict(kw.get("frontmatter") or {}), links=extract_links(body),
        )
        if errors := validate(c):
            raise ToolError("; ".join(errors))
        return c

    async def create(self, path: str, **kw: Any) -> dict[str, Any]:
        version = self._c.create(self._t, self._concept(path, **kw), self._actor)
        if version is None:
            raise ToolError(f"concept already exists at {path}")
        return {"path": path, "version": version}

    async def update(
        self, path: str, expected_version: int | None = None, **kw: Any
    ) -> dict[str, Any]:
        existing = self._c.read(self._t, path)
        if existing is None:
            raise ToolError(f"no concept at {path}")
        merged = {"type": existing.type, "title": existing.title,
                  "description": existing.description, "body": existing.body} | kw
        result = self._c.update(
            self._t, self._concept(path, **merged), self._actor, expected_version
        )
        if isinstance(result, Conflict):
            return {"conflict": True, "current_version": result.current_version,
                    "current_body": result.current_body}
        return {"path": path, "version": result}

    async def search(self, query: str, limit: int, prefix: str | None = None) -> list[dict]:
        return [
            {"path": h.path, "type": h.type, "title": h.title,
             "description": h.description, "score": h.score}
            for h in self._c.search(self._t, query, limit, prefix)
        ]

    async def grep(self, pattern: str, limit: int) -> list[dict]:
        return [{"path": p, "snippet": s} for p, s in self._c.grep(self._t, pattern, limit)]

    async def list_(self, prefix: str = "") -> dict[str, Any]:
        rows = self._c.list_(self._t, prefix)
        counts: dict[str, int] = {}
        for _, type_ in rows:
            counts[type_] = counts.get(type_, 0) + 1
        return {"paths": [p for p, _ in rows], "counts": counts}

    async def read(self, path: str) -> dict[str, Any] | None:
        c = self._c.read(self._t, path)
        if c is None:
            return None
        return {"path": c.path, "type": c.type, "title": c.title,
                "description": c.description, "body": c.body, "version": c.version,
                "links": list(c.links), "backlinks": self._c.backlinks(self._t, path)}

    async def relate(self, from_path: str, to_path: str) -> dict[str, Any]:
        source = self._c.read(self._t, from_path)
        if source is None:
            raise ToolError(f"no concept at {from_path}")
        body = f"{source.body}\n\n[{to_path}]({to_path}.md)\n"
        return await self.update(from_path, expected_version=source.version, body=body)
```

```python
"""Bundle import and export. Generated index and log files are never concepts."""

from __future__ import annotations

from pathlib import Path
from uuid import UUID

from okf_core import parse, serialize
from keepsake.store.concepts import ConceptStore

_GENERATED = {"index", "log"}


def import_bundle(concepts: ConceptStore, tenant_id: UUID, root: Path) -> int:
    count = 0
    for file in sorted(root.rglob("*.md")):
        path = str(file.relative_to(root).with_suffix(""))
        if Path(path).name in _GENERATED:
            continue
        concept = parse(file.read_text(), path)
        if concepts.create(tenant_id, concept, "import") is None:
            existing = concepts.read(tenant_id, path)
            assert existing is not None
            concepts.update(tenant_id, concept, "import", existing.version)
        count += 1
    return count


def export_bundle(concepts: ConceptStore, tenant_id: UUID, root: Path) -> int:
    root.mkdir(parents=True, exist_ok=True)
    paths = [p for p, _ in concepts.list_(tenant_id, "")]
    for path in paths:
        concept = concepts.read(tenant_id, path)
        assert concept is not None
        target = root / f"{path}.md"
        target.parent.mkdir(parents=True, exist_ok=True)
        target.write_text(serialize(concept))
    (root / "index.md").write_text(_render_index(paths))
    (root / "log.md").write_text(_render_log(concepts, tenant_id))
    return len(paths)
```

`_render_index` groups paths by their first segment. `_render_log` reads
`concept_revision` in `created_at` order and renders dated entries.

### Task 12: kind end-to-end test

**Files:**
- Create: `e2e/kind.yaml`, `e2e/run.sh`, `e2e/postgres.yaml`, `e2e/test_deployed.py`
- Create: `.github/workflows/e2e.yaml`

**Interfaces:**
- Consumes: the chart from Task 11.
- Produces: a single command that stands up a cluster, installs the chart, asserts behaviour, and tears down.

**Why this exists.** Three properties cannot be proven by unit tests:

1. `verify` actually crash-loops the pod when the app role is privileged. The whole isolation story rests on that check firing in a real deployment, not on a unit test calling it directly.
2. The migration Job completes before the server pod starts.
3. The MCP endpoint is reachable and advertises the right tools over real HTTP.

The e2e uses `postgres.mode=existing` against a plain Postgres Deployment. That is
deliberate: the assertions above need *a* Postgres, not CloudNativePG specifically, and
skipping the operator install keeps the suite fast and stable. Managed mode is exercised by
a separate, slower CI job.

```python
import json, subprocess, urllib.request
import pytest

BASE = "http://localhost:30800"


def _kubectl(*args: str) -> str:
    return subprocess.run(["kubectl", *args], capture_output=True, text=True,
                          check=True).stdout


def _rpc(method: str, params: dict | None = None) -> dict:
    payload = json.dumps(
        {"jsonrpc": "2.0", "id": 1, "method": method, "params": params or {}}
    ).encode()
    req = urllib.request.Request(
        f"{BASE}/mcp", data=payload,
        headers={"Content-Type": "application/json", "Accept": "application/json"},
    )
    with urllib.request.urlopen(req, timeout=10) as r:
        return json.loads(r.read())


def test_migration_job_succeeded():
    assert "Complete" in _kubectl("get", "job", "-l", "app=keepsake-migrate",
                                  "-o", "jsonpath={.items[0].status.conditions[*].type}")


def test_server_pod_is_ready():
    assert _kubectl("get", "pods", "-l", "app=keepsake",
                    "-o", "jsonpath={.items[0].status.phase}") == "Running"


def test_mcp_endpoint_advertises_the_tool_surface():
    names = {t["name"] for t in _rpc("tools/list")["result"]["tools"]}
    assert names == {"okf_list", "okf_search", "okf_grep", "okf_read",
                     "okf_create", "okf_update", "okf_relate"}


def test_search_and_grep_require_a_limit():
    tools = {t["name"]: t for t in _rpc("tools/list")["result"]["tools"]}
    for name in ("okf_search", "okf_grep"):
        assert "limit" in tools[name]["inputSchema"]["required"]


def test_a_concept_round_trips_through_the_deployed_server():
    _rpc("tools/call", {"name": "okf_create", "arguments": {
        "path": "e2e/smoke", "type": "Concept", "title": "Smoke",
        "description": "", "body": "deployed", "links": []}})
    hits = _rpc("tools/call", {"name": "okf_search",
                               "arguments": {"query": "smoke", "limit": 5}})
    assert any(h["path"] == "e2e/smoke" for h in hits["result"]["structuredContent"])


def test_privileged_app_role_crash_loops_the_pod():
    """The isolation story depends on verify() firing in a real deployment."""
    subprocess.run(
        ["helm", "upgrade", "keepsake-bad", "charts/keepsake",
         "--set", "postgres.mode=existing",
         "--set", "postgres.dsn=postgres://postgres:postgres@postgres:5432/keepsake",
         "--install", "--wait", "--timeout", "60s"],
        capture_output=True, text=True,
    )
    logs = subprocess.run(
        ["kubectl", "logs", "-l", "app=keepsake-bad", "--tail=20"],
        capture_output=True, text=True,
    ).stdout
    assert "must not connect as a superuser" in logs
```
